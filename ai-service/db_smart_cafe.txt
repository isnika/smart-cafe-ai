-- ============================================================================
-- SMART CAFE - DATABASE SCRIPT (MySQL 8.0.17+; CHECK constraints are enforced, multi-valued JSON index is used)
-- ============================================================================
-- Architecture: database-per-service. No cross-service foreign keys; cross-service
-- columns are "external references" kept consistent through Outbox/Inbox events.
--
-- 01. Identity Service      -> smart_cafe_identity_db
-- 02. Catalog Service       -> smart_cafe_catalog_db
-- 03. Store Service         -> smart_cafe_store_db      (includes Workforce)
-- 04. Order Service         -> smart_cafe_order_db
-- 05. Payment Service       -> smart_cafe_payment_db
-- 06. Inventory Service     -> smart_cafe_inventory_db
-- 07. Engagement Service    -> smart_cafe_engagement_db
-- 08. Intelligence Service  -> smart_cafe_intelligence_db
--
-- GLOBAL CONVENTIONS
-- 1. All DATETIME values are stored in UTC. Configure every connection with
--    time_zone = '+00:00'. Branch-local time uses Branches.timezone for reporting.
-- 2. Money is VND, stored as DECIMAL(15,0) (no fractional part).
-- 3. Prices are VAT-INCLUSIVE. Orders.vat_amount is the VAT portion contained in
--    final_amount (informational), it is NOT added on top.
--    Orders.final_amount = sub_total - item_discount_amount
--                          - voucher_discount_amount + delivery_fee
-- 4. Cross-service actor references ALWAYS use Identity Users.id (user_id).
--    Only Store Service uses Staff_Profiles.id internally.
-- 5. Never hard-delete business master data (users, products, branches...).
--    Use status columns. FKs use RESTRICT unless the child is pure detail data.
-- 6. Rules that cannot be expressed as a constraint are written as "APP RULE"
--    in comments and MUST be enforced by the service inside one transaction.
-- ============================================================================

SET NAMES utf8mb4;

-- ============================================================================
-- 01. Identity Service
-- ============================================================================
CREATE DATABASE IF NOT EXISTS smart_cafe_identity_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE smart_cafe_identity_db;

CREATE TABLE Roles (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE,
    description VARCHAR(255)
);

-- Branch scope of staff is owned by Store Service (Staff_Profiles.branch_id) and
-- should be placed in the access token claims; it is intentionally not duplicated here.
CREATE TABLE Users (
    id VARCHAR(36) PRIMARY KEY,
    email VARCHAR(100) NULL UNIQUE,
    phone VARCHAR(20) NULL UNIQUE,
    password_hash VARCHAR(255) NULL,
    google_id VARCHAR(100) NULL UNIQUE,
    full_name VARCHAR(100) NOT NULL,
    avatar VARCHAR(255),
    role_id INT NOT NULL,
    status ENUM('Unverified', 'Active', 'Banned', 'Deactivated') NOT NULL DEFAULT 'Unverified'
        COMMENT 'Deactivated is set when a staff member is terminated (Store event) or the user closes the account',
    email_verified_at DATETIME NULL,
    phone_verified_at DATETIME NULL,
    last_login_at DATETIME NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_users_role FOREIGN KEY (role_id) REFERENCES Roles(id),
    CONSTRAINT chk_users_identifier CHECK (email IS NOT NULL OR phone IS NOT NULL OR google_id IS NOT NULL),
    CONSTRAINT chk_users_credential CHECK (password_hash IS NOT NULL OR google_id IS NOT NULL),
    INDEX idx_users_status (status),
    INDEX idx_users_role (role_id)
);

CREATE TABLE Refresh_Tokens (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL,
    token_hash VARCHAR(255) NOT NULL UNIQUE COMMENT 'Store only a hash of the refresh token',
    device_info VARCHAR(255) NULL,
    ip_address VARCHAR(45) NULL,
    expires_at DATETIME NOT NULL,
    revoked_at DATETIME NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_refresh_tokens_user FOREIGN KEY (user_id) REFERENCES Users(id) ON DELETE CASCADE,
    INDEX idx_refresh_user (user_id, revoked_at),
    INDEX idx_refresh_expiry (expires_at)
);

CREATE TABLE OTP_Logs (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id VARCHAR(36) NULL COMMENT 'NULL for Register before the user exists',
    channel ENUM('Email', 'Phone') NOT NULL,
    target VARCHAR(100) NOT NULL COMMENT 'Email address or phone number matching channel',
    otp_hash VARCHAR(255) NOT NULL COMMENT 'One-way hash only; never store the plaintext OTP',
    purpose ENUM('Register', 'Reset_Password', 'Change_Password', 'Change_Email', 'Change_Phone') NOT NULL,
    expires_at DATETIME NOT NULL,
    attempt_count INT UNSIGNED NOT NULL DEFAULT 0,
    max_attempts INT UNSIGNED NOT NULL DEFAULT 5,
    is_used BOOLEAN NOT NULL DEFAULT FALSE,
    used_at DATETIME NULL,
    ip_address VARCHAR(45) NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_otp_max_attempts CHECK (max_attempts > 0),
    CONSTRAINT chk_otp_attempts CHECK (attempt_count <= max_attempts),
    CONSTRAINT chk_otp_used CHECK ((is_used = FALSE AND used_at IS NULL) OR (is_used = TRUE AND used_at IS NOT NULL)),
    -- APP RULE: rate-limit resend by counting rows per (target, purpose) in a recent window.
    INDEX idx_otp_target_purpose_created (target, purpose, created_at),
    INDEX idx_otp_expiry_used (expires_at, is_used)
);

CREATE TABLE Addresses (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL,
    label VARCHAR(50) NULL COMMENT 'Home, Office...',
    receiver_name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    province VARCHAR(100) NOT NULL,
    district VARCHAR(100) NULL COMMENT 'Legacy level; Vietnam removed district level from 2025-07-01',
    ward VARCHAR(100) NOT NULL,
    address_line VARCHAR(255) NOT NULL,
    latitude DECIMAL(10,7) NULL,
    longitude DECIMAL(10,7) NULL,
    is_default BOOLEAN NOT NULL DEFAULT FALSE,
    default_user_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN is_default THEN user_id ELSE NULL END) STORED
        COMMENT 'Used only to enforce one default address per user',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_addresses_user FOREIGN KEY (user_id) REFERENCES Users(id) ON DELETE RESTRICT, -- RESTRICT: user_id is the base column of a STORED generated column, so MySQL forbids CASCADE here
    CONSTRAINT chk_addresses_geo CHECK (
        (latitude IS NULL AND longitude IS NULL)
        OR (latitude BETWEEN -90 AND 90 AND longitude BETWEEN -180 AND 180)
    ),
    UNIQUE KEY uq_one_default_address_per_user (default_user_id),
    INDEX idx_addresses_user (user_id)
);

-- Events published: identity.user.banned, identity.user.deactivated.
-- Events consumed: store.staff.terminated (-> Users.status = 'Deactivated').
CREATE TABLE Outbox_Events (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    event_id VARCHAR(36) NOT NULL UNIQUE,
    aggregate_type VARCHAR(50) NOT NULL,
    aggregate_id VARCHAR(64) NOT NULL,
    event_type VARCHAR(100) NOT NULL,
    event_version SMALLINT UNSIGNED NOT NULL DEFAULT 1,
    correlation_id VARCHAR(64) NULL,
    payload JSON NOT NULL,
    status ENUM('Pending', 'Published', 'Failed') NOT NULL DEFAULT 'Pending',
    retry_count INT UNSIGNED NOT NULL DEFAULT 0,
    next_retry_at DATETIME NULL,
    last_error VARCHAR(500) NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    published_at DATETIME NULL,
    INDEX idx_identity_outbox_dispatch (status, next_retry_at, created_at),
    INDEX idx_identity_outbox_aggregate (aggregate_type, aggregate_id)
);

CREATE TABLE Processed_Events (
    event_id VARCHAR(36) PRIMARY KEY,
    event_type VARCHAR(100) NOT NULL,
    source_service VARCHAR(100) NULL,
    processed_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_identity_processed_type_time (event_type, processed_at)
);

CREATE TABLE Audit_Logs (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    actor_user_id VARCHAR(36) NULL COMMENT 'External reference to Identity Service; NULL for system actions',
    action VARCHAR(100) NOT NULL,
    entity_type VARCHAR(50) NOT NULL,
    entity_id VARCHAR(64) NOT NULL,
    old_value JSON NULL,
    new_value JSON NULL,
    ip_address VARCHAR(45) NULL,
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_identity_audit_entity (entity_type, entity_id, created_at),
    INDEX idx_identity_audit_actor (actor_user_id, created_at)
);

-- ============================================================================
-- 02. Catalog Service
-- ============================================================================
-- AVAILABILITY RULE: an item is sellable at a branch only when
--   Products.status = 'Active' AND Product_Variants.status = 'Active'
--   AND (no Branch_Product_Availability row exists OR is_available = TRUE).
-- "No row" means AVAILABLE. Options follow the same rule.
CREATE DATABASE IF NOT EXISTS smart_cafe_catalog_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE smart_cafe_catalog_db;

CREATE TABLE Categories (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    image VARCHAR(255),
    sort_order INT NOT NULL DEFAULT 0,
    status ENUM('Active', 'Inactive') NOT NULL DEFAULT 'Active'
);

CREATE TABLE Products (
    id INT AUTO_INCREMENT PRIMARY KEY,
    category_id INT NOT NULL,
    name VARCHAR(100) NOT NULL,
    image VARCHAR(255) NULL,
    description TEXT,
    temperature ENUM('Hot', 'Iced', 'Blended', 'Hot_Or_Iced') NULL
        COMMENT 'Serving temperature; NULL for non-drink items (food, merchandise)',
    caffeine_level ENUM('None', 'Low', 'Medium', 'High') NOT NULL DEFAULT 'None',
    flavor_tags JSON NULL COMMENT 'JSON array of lowercase tags, e.g. ["sweet","creamy","nutty"]; used for search and AI recommendations',
    sort_order INT NOT NULL DEFAULT 0,
    status ENUM('Active', 'Inactive') NOT NULL DEFAULT 'Active',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_products_category FOREIGN KEY (category_id) REFERENCES Categories(id),
    CONSTRAINT chk_products_flavor_tags CHECK (flavor_tags IS NULL OR JSON_TYPE(flavor_tags) = 'ARRAY'),
    INDEX idx_products_category_status (category_id, status, sort_order),
    INDEX idx_products_temperature_caffeine (temperature, caffeine_level),
    INDEX idx_products_flavor_tags ((CAST(flavor_tags -> '$[*]' AS CHAR(30) ARRAY)))
);

CREATE TABLE Product_Variants (
    id INT AUTO_INCREMENT PRIMARY KEY,
    product_id INT NOT NULL,
    variant_name VARCHAR(50) NOT NULL DEFAULT 'Standard',
    price DECIMAL(15,0) NOT NULL,
    sort_order INT NOT NULL DEFAULT 0,
    status ENUM('Active', 'Inactive') NOT NULL DEFAULT 'Active' COMMENT 'Global menu lifecycle status; branch availability is stored separately',
    CONSTRAINT fk_variants_product FOREIGN KEY (product_id) REFERENCES Products(id) ON DELETE RESTRICT,
    CONSTRAINT chk_variants_price CHECK (price >= 0),
    UNIQUE KEY uq_variant_per_product (product_id, variant_name)
);

CREATE TABLE Variant_Price_History (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    variant_id INT NOT NULL,
    old_price DECIMAL(15,0) NULL,
    new_price DECIMAL(15,0) NOT NULL,
    changed_by VARCHAR(36) NULL COMMENT 'External reference to Identity Service',
    changed_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_price_history_variant FOREIGN KEY (variant_id) REFERENCES Product_Variants(id) ON DELETE RESTRICT,
    INDEX idx_price_history_variant_time (variant_id, changed_at)
);

CREATE TABLE Branch_Product_Availability (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL COMMENT 'External reference to Store Service; no cross-service FK',
    variant_id INT NOT NULL,
    is_available BOOLEAN NOT NULL DEFAULT TRUE,
    unavailable_reason VARCHAR(255) NULL,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_branch_product_availability_variant FOREIGN KEY (variant_id) REFERENCES Product_Variants(id) ON DELETE RESTRICT,
    UNIQUE KEY uq_branch_variant_availability (branch_id, variant_id),
    INDEX idx_branch_product_availability_branch (branch_id, is_available),
    INDEX idx_branch_product_availability_variant (variant_id)
);

CREATE TABLE Product_Option_Groups (
    id INT AUTO_INCREMENT PRIMARY KEY,
    product_id INT NOT NULL,
    name VARCHAR(50) NOT NULL,
    selection_type ENUM('Single', 'Multiple') NOT NULL,
    is_required BOOLEAN NOT NULL DEFAULT FALSE,
    min_select TINYINT UNSIGNED NOT NULL DEFAULT 0,
    max_select TINYINT UNSIGNED NOT NULL DEFAULT 1,
    sort_order INT NOT NULL DEFAULT 0,
    CONSTRAINT fk_option_groups_product FOREIGN KEY (product_id) REFERENCES Products(id) ON DELETE RESTRICT,
    CONSTRAINT chk_option_groups_range CHECK (max_select >= 1 AND min_select <= max_select),
    CONSTRAINT chk_option_groups_single CHECK (selection_type <> 'Single' OR max_select = 1),
    CONSTRAINT chk_option_groups_required CHECK (is_required = FALSE OR min_select >= 1),
    UNIQUE KEY uq_option_group_per_product (product_id, name)
);

CREATE TABLE Product_Options (
    id INT AUTO_INCREMENT PRIMARY KEY,
    group_id INT NOT NULL,
    option_name VARCHAR(100) NOT NULL,
    extra_price DECIMAL(15,0) NOT NULL DEFAULT 0,
    is_default BOOLEAN NOT NULL DEFAULT FALSE,
    sort_order INT NOT NULL DEFAULT 0,
    status ENUM('Active', 'Inactive') NOT NULL DEFAULT 'Active',
    CONSTRAINT fk_options_group FOREIGN KEY (group_id) REFERENCES Product_Option_Groups(id) ON DELETE RESTRICT,
    CONSTRAINT chk_options_price CHECK (extra_price >= 0),
    UNIQUE KEY uq_option_per_group (group_id, option_name)
);

CREATE TABLE Branch_Option_Availability (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL COMMENT 'External reference to Store Service; no cross-service FK',
    option_id INT NOT NULL,
    is_available BOOLEAN NOT NULL DEFAULT TRUE,
    unavailable_reason VARCHAR(255) NULL,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_branch_option_availability_option FOREIGN KEY (option_id) REFERENCES Product_Options(id) ON DELETE RESTRICT,
    UNIQUE KEY uq_branch_option_availability (branch_id, option_id),
    INDEX idx_branch_option_availability_branch (branch_id, is_available),
    INDEX idx_branch_option_availability_option (option_id)
);

-- Product-level price for display. The source of truth stays Product_Variants.price
-- (a product can have several sizes with different prices), so there is no duplicate column to drift.
-- Search by flavor: WHERE 'creamy' MEMBER OF (flavor_tags)
CREATE VIEW v_products_full AS
SELECT p.id, p.category_id, p.name, p.image, p.description,
       p.temperature, p.caffeine_level, p.flavor_tags, p.status, p.sort_order,
       MIN(v.price) AS price,
       MAX(v.price) AS max_price
FROM Products p
LEFT JOIN Product_Variants v ON v.product_id = p.id AND v.status = 'Active'
GROUP BY p.id, p.category_id, p.name, p.image, p.description,
         p.temperature, p.caffeine_level, p.flavor_tags, p.status, p.sort_order;

-- ============================================================================
-- 03. Store Service (includes Workforce capability)
-- ============================================================================
CREATE DATABASE IF NOT EXISTS smart_cafe_store_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE smart_cafe_store_db;

CREATE TABLE Branches (
    id INT AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(20) NOT NULL UNIQUE,
    name VARCHAR(100) NOT NULL,
    address VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    --latitude DECIMAL(10,7) NULL,
	--longitude DECIMAL(10,7) NULL, 
    timezone VARCHAR(50) NOT NULL DEFAULT 'Asia/Ho_Chi_Minh',
    opening_time TIME NULL,
    closing_time TIME NULL,
    accepts_online_orders BOOLEAN NOT NULL DEFAULT TRUE,
    delivery_radius_km DECIMAL(5,2) NULL COMMENT 'NULL = delivery not offered',
    delivery_fee DECIMAL(15,0) NOT NULL DEFAULT 0,
    status ENUM('Active', 'Inactive') NOT NULL DEFAULT 'Active',
    CONSTRAINT chk_branches_fee CHECK (delivery_fee >= 0)
);

-- Table status is DERIVED from open sessions (see view v_table_status); it is not stored.
CREATE TABLE Tables (
    id INT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL,
    table_number VARCHAR(20) NOT NULL,
    qr_code VARCHAR(255) UNIQUE,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    UNIQUE KEY uq_table_number_per_branch (branch_id, table_number),
    CONSTRAINT fk_tables_branch FOREIGN KEY (branch_id) REFERENCES Branches(id) ON DELETE RESTRICT
);

CREATE TABLE Table_Sessions (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    table_id INT NOT NULL,
    session_code VARCHAR(50) NOT NULL UNIQUE COMMENT 'Must be a random, unguessable token (e.g. 128-bit)',
    customer_id VARCHAR(36) NULL COMMENT 'External reference to Identity Service; guest QR session may be NULL',
    status ENUM('Active', 'Payment_Pending', 'Completed', 'Cancelled') NOT NULL DEFAULT 'Active',
    opened_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    expires_at DATETIME NULL COMMENT 'Idle expiry to stop reuse of an old QR session',
    closed_at DATETIME NULL,
    open_table_id INT GENERATED ALWAYS AS (
        CASE WHEN status IN ('Active', 'Payment_Pending') THEN table_id ELSE NULL END
    ) STORED COMMENT 'Used only to enforce one open session per table',
    CONSTRAINT fk_table_sessions_table FOREIGN KEY (table_id) REFERENCES Tables(id) ON DELETE RESTRICT,
    CONSTRAINT chk_table_sessions_closed CHECK (
        (status IN ('Active', 'Payment_Pending') AND closed_at IS NULL)
        OR (status IN ('Completed', 'Cancelled') AND closed_at IS NOT NULL)
    ),
    UNIQUE KEY uq_one_open_session_per_table (open_table_id),
    INDEX idx_table_sessions_customer (customer_id),
    INDEX idx_table_sessions_table_status (table_id, status)
);

CREATE VIEW v_table_status AS
SELECT t.id AS table_id, t.branch_id, t.table_number, t.is_active,
       CASE WHEN ts.id IS NULL THEN 'Empty' ELSE 'Occupied' END AS status,
       ts.id AS open_session_id
FROM Tables t
LEFT JOIN Table_Sessions ts ON ts.open_table_id = t.id;

-- Workforce capability. Authentication stays in Identity; user_id is an external reference.
CREATE TABLE Staff_Profiles (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL COMMENT 'External reference to Identity Service; one user maps to at most one staff profile',
    branch_id INT NOT NULL,
    employee_code VARCHAR(30) NOT NULL,
    job_title VARCHAR(100) NOT NULL COMMENT 'Barista, Cashier, Shift Leader, Store Manager...',
    employment_type ENUM('Full_Time', 'Part_Time', 'Probation', 'Intern') NOT NULL DEFAULT 'Full_Time',
    hire_date DATE NOT NULL,
    termination_date DATE NULL,
    status ENUM('Active', 'On_Leave', 'Inactive', 'Terminated') NOT NULL DEFAULT 'Active',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_staff_profiles_branch FOREIGN KEY (branch_id) REFERENCES Branches(id) ON DELETE RESTRICT,
    CONSTRAINT chk_staff_termination CHECK (termination_date IS NULL OR termination_date >= hire_date),
    CONSTRAINT chk_staff_terminated CHECK (status <> 'Terminated' OR termination_date IS NOT NULL),
    UNIQUE KEY uq_staff_profile_user (user_id),
    UNIQUE KEY uq_staff_employee_code (employee_code),
    UNIQUE KEY uq_staff_id_branch (id, branch_id) COMMENT 'Target of composite FK from Schedules',
    INDEX idx_staff_branch_status (branch_id, status),
    INDEX idx_staff_job_title (job_title)
);
-- APP RULE: on Terminated, publish store.staff.terminated so Identity sets Users.status = 'Deactivated'.
-- APP RULE: to transfer a staff member, delete their future Schedules first (composite FK blocks the change),
--           then update branch_id and insert a Staff_Branch_History row in the same transaction.

CREATE TABLE Staff_Branch_History (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    staff_id BIGINT NOT NULL,
    from_branch_id INT NULL,
    to_branch_id INT NOT NULL,
    effective_date DATE NOT NULL,
    changed_by VARCHAR(36) NULL COMMENT 'External reference to Identity Service',
    note VARCHAR(255) NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_staff_history_staff FOREIGN KEY (staff_id) REFERENCES Staff_Profiles(id) ON DELETE RESTRICT,
    CONSTRAINT fk_staff_history_from FOREIGN KEY (from_branch_id) REFERENCES Branches(id) ON DELETE RESTRICT,
    CONSTRAINT fk_staff_history_to FOREIGN KEY (to_branch_id) REFERENCES Branches(id) ON DELETE RESTRICT,
    INDEX idx_staff_history_staff (staff_id, effective_date)
);

CREATE TABLE Shifts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL,
    name VARCHAR(50) NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    is_overnight BOOLEAN GENERATED ALWAYS AS (end_time < start_time) STORED COMMENT 'TRUE when the shift ends on the next day',
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT fk_shifts_branch FOREIGN KEY (branch_id) REFERENCES Branches(id) ON DELETE RESTRICT,
    CONSTRAINT chk_shifts_time CHECK (start_time <> end_time),
    UNIQUE KEY uq_shift_per_branch (branch_id, name),
    UNIQUE KEY uq_shift_id_branch (id, branch_id) COMMENT 'Target of composite FK from Schedules'
);

-- One schedule row per staff per day (prevents double booking / overlapping shifts).
-- To support split shifts later, drop uq_staff_work_date and add an overlap check in the service.
CREATE TABLE Schedules (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL,
    staff_id BIGINT NOT NULL,
    shift_id INT NOT NULL,
    work_date DATE NOT NULL,
    status ENUM('Scheduled', 'Completed', 'Absent', 'Leave') NOT NULL DEFAULT 'Scheduled',
    CONSTRAINT fk_schedules_staff FOREIGN KEY (staff_id, branch_id) REFERENCES Staff_Profiles(id, branch_id) ON DELETE RESTRICT,
    CONSTRAINT fk_schedules_shift FOREIGN KEY (shift_id, branch_id) REFERENCES Shifts(id, branch_id) ON DELETE RESTRICT,
    UNIQUE KEY uq_staff_work_date (staff_id, work_date),
    INDEX idx_schedules_branch_date (branch_id, work_date),
    INDEX idx_schedules_shift_date (shift_id, work_date)
);

CREATE TABLE Timekeeping (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    schedule_id BIGINT NOT NULL,
    check_in DATETIME NULL,
    check_out DATETIME NULL,
    source ENUM('Manual', 'Device', 'GPS', 'QR') NOT NULL DEFAULT 'Manual',
    is_adjusted BOOLEAN NOT NULL DEFAULT FALSE,
    adjusted_by_staff_id BIGINT NULL,
    adjustment_reason VARCHAR(255) NULL,
    CONSTRAINT fk_timekeeping_schedule FOREIGN KEY (schedule_id) REFERENCES Schedules(id) ON DELETE RESTRICT,
    CONSTRAINT fk_timekeeping_adjuster FOREIGN KEY (adjusted_by_staff_id) REFERENCES Staff_Profiles(id) ON DELETE RESTRICT,
    CONSTRAINT chk_timekeeping_order CHECK (check_out IS NULL OR check_in IS NULL OR check_out >= check_in),
    CONSTRAINT chk_timekeeping_adjust CHECK (
        (is_adjusted = FALSE)
        OR (is_adjusted = TRUE AND adjusted_by_staff_id IS NOT NULL AND adjustment_reason IS NOT NULL)
    ),
    UNIQUE KEY uq_timekeeping_schedule (schedule_id)
);

CREATE TABLE Leave_Requests (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    staff_id BIGINT NOT NULL,
    leave_type ENUM('Annual', 'Sick', 'Unpaid', 'Other') NOT NULL DEFAULT 'Annual',
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    reason TEXT NOT NULL,

-- 04. Order Service
-- ============================================================================
-- ORDER STATE MACHINE (enforced in the service):
--   Online/Delivery : Pending_Payment -> Pending -> Confirmed -> Preparing -> Ready -> Shipping -> Completed
--   Online/Pickup   : Pending_Payment -> Pending -> Confirmed -> Preparing -> Ready -> Completed
--   QR/POS Dine_In  : Pending -> Confirmed -> Preparing -> Ready -> Served -> Completed
--   Any open state  -> Cancelled (by customer/staff) or Rejected (by branch, before Preparing)
-- Pending_Payment = waiting for VNPay/Momo; auto-cancel when expires_at passes.
-- payment_status is a read-model updated from Payment Service events.
CREATE DATABASE IF NOT EXISTS smart_cafe_order_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE smart_cafe_order_db;

CREATE TABLE Orders (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    order_code VARCHAR(20) NOT NULL UNIQUE,
    idempotency_key VARCHAR(64) NULL UNIQUE COMMENT 'Prevents duplicate order creation on retry/double-submit',
    user_id VARCHAR(36) NULL COMMENT 'External reference to Identity Service; required for Online orders',
    created_by VARCHAR(36) NULL COMMENT 'External reference to Identity Service; staff user for POS orders',
    branch_id INT NOT NULL COMMENT 'External reference to Store Service',
    table_session_id BIGINT NULL COMMENT 'External reference to Store Service; required for Dine_In',
    order_source ENUM('Online', 'QR', 'POS') NOT NULL,
    fulfillment_type ENUM('Delivery', 'Pickup', 'Dine_In') NOT NULL,
    delivery_name VARCHAR(100) NULL,
    delivery_phone VARCHAR(20) NULL,
    delivery_address VARCHAR(255) NULL,
    delivery_latitude DECIMAL(10,7) NULL,
    delivery_longitude DECIMAL(10,7) NULL,

    item_discount_amount DECIMAL(15,0) NOT NULL DEFAULT 0 COMMENT 'Sum of Order_Items.discount_amount (e.g. loyalty reward)',

    expires_at DATETIME NULL COMMENT 'Deadline for Pending_Payment orders',

    cancelled_at DATETIME NULL,
    completed_at DATETIME NULL,
    note VARCHAR(255) NULL,
    version INT UNSIGNED NOT NULL DEFAULT 0 COMMENT 'Optimistic lock: UPDATE ... WHERE id=? AND version=?',

        AND item_discount_amount + voucher_discount_amount <= sub_total
    ),
    CONSTRAINT chk_orders_final_formula CHECK (
        final_amount = sub_total - item_discount_amount - voucher_discount_amount + delivery_fee
    ),
    CONSTRAINT chk_orders_source_fulfillment CHECK (
        (order_source = 'QR'  AND fulfillment_type = 'Dine_In')
        OR (order_source = 'POS'    AND fulfillment_type IN ('Dine_In', 'Pickup'))
        OR (order_source = 'Online' AND fulfillment_type IN ('Delivery', 'Pickup'))
    ),
    CONSTRAINT chk_orders_table_session CHECK (
        (fulfillment_type = 'Dine_In' AND table_session_id IS NOT NULL)
        OR (fulfillment_type <> 'Dine_In' AND table_session_id IS NULL)
    ),
    CONSTRAINT chk_orders_delivery CHECK (
        (fulfillment_type = 'Delivery' AND delivery_name IS NOT NULL AND delivery_phone IS NOT NULL AND delivery_address IS NOT NULL)
        OR (fulfillment_type <> 'Delivery' AND delivery_fee = 0)

    CONSTRAINT chk_orders_online_user CHECK (order_source <> 'Online' OR user_id IS NOT NULL),
    CONSTRAINT chk_orders_pos_staff CHECK (order_source <> 'POS' OR created_by IS NOT NULL),
    CONSTRAINT chk_orders_cancel_reason CHECK (status NOT IN ('Cancelled', 'Rejected') OR cancel_reason IS NOT NULL),
    CONSTRAINT chk_orders_completed_at CHECK (status <> 'Completed' OR completed_at IS NOT NULL),
    INDEX idx_orders_user_created (user_id, created_at),

-- ============================================================================
-- 05. Payment Service
-- ============================================================================
-- MODEL: every payment is linked to orders ONLY through Payment_Order_Allocations
-- (single-order payment = 1 allocation; table-session settlement = N allocations).
-- APP RULES (one transaction, lock the order rows with SELECT ... FOR UPDATE):
--   * SUM(allocated_amount) of a payment = Payments.amount
--   * SUM(allocations of Success payments) - SUM(refund allocations) <= Orders.final_amount, per order
--   * Cash/COD: cash is "owned" by the cashier shift that received/remitted it.
CREATE DATABASE IF NOT EXISTS smart_cafe_payment_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE smart_cafe_payment_db;

CREATE TABLE Cashier_Shifts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL COMMENT 'External reference to Store Service',
    register_code VARCHAR(20) NOT NULL DEFAULT 'MAIN' COMMENT 'Cash register / drawer identifier',
    staff_user_id VARCHAR(36) NOT NULL COMMENT 'External reference to Identity Service (Users.id)',
    opening_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    closing_time DATETIME NULL,
    opening_cash DECIMAL(15,0) NOT NULL DEFAULT 0,
    -- Totals below are computed and frozen when the shift is closed.
    total_orders INT NOT NULL DEFAULT 0,
    total_cash_sales DECIMAL(15,0) NOT NULL DEFAULT 0,
    total_online_sales DECIMAL(15,0) NOT NULL DEFAULT 0,
    total_cash_refunds DECIMAL(15,0) NOT NULL DEFAULT 0,
    total_cash_in DECIMAL(15,0) NOT NULL DEFAULT 0,
    total_cash_out DECIMAL(15,0) NOT NULL DEFAULT 0,
    expected_cash DECIMAL(15,0) NULL COMMENT 'opening + cash sales - cash refunds + cash in - cash out',
    closing_cash DECIMAL(15,0) NULL COMMENT 'Counted cash at close',
    discrepancy DECIMAL(15,0) GENERATED ALWAYS AS (closing_cash - expected_cash) STORED,
    status ENUM('Open', 'Closed') NOT NULL DEFAULT 'Open',
    open_staff_user_id VARCHAR(36) GENERATED ALWAYS AS (CASE WHEN status = 'Open' THEN staff_user_id ELSE NULL END) STORED,
    open_register_key VARCHAR(60) GENERATED ALWAYS AS (CASE WHEN status = 'Open' THEN CONCAT(branch_id, ':', register_code) ELSE NULL END) STORED,
    CONSTRAINT chk_cashier_shift_amounts CHECK (
        opening_cash >= 0 AND total_orders >= 0 AND total_cash_sales >= 0 AND total_online_sales >= 0
        AND total_cash_refunds >= 0 AND total_cash_in >= 0 AND total_cash_out >= 0
    ),
    CONSTRAINT chk_cashier_shift_closed CHECK (
        (status = 'Open' AND closing_time IS NULL AND closing_cash IS NULL AND expected_cash IS NULL)
        OR (status = 'Closed' AND closing_time IS NOT NULL AND closing_cash IS NOT NULL AND expected_cash IS NOT NULL)
    ),
    UNIQUE KEY uq_one_open_shift_per_staff (open_staff_user_id),
    UNIQUE KEY uq_one_open_shift_per_register (open_register_key),
    INDEX idx_cashier_shift_branch_status (branch_id, status),
    INDEX idx_cashier_shift_staff (staff_user_id)
);

    amount DECIMAL(15,0) NOT NULL,
    reason VARCHAR(255) NOT NULL,
    created_by VARCHAR(36) NOT NULL COMMENT 'External reference to Identity Service',
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    INDEX idx_cash_movements_shift (cashier_shift_id, created_at)

CREATE TABLE Payments (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL COMMENT 'External reference to Store Service',
    cashier_shift_id INT NULL COMMENT 'Shift that received the cash (Cash); for COD set when the shipper remits the cash',
    idempotency_key VARCHAR(64) NULL UNIQUE,
    payment_method ENUM('Cash', 'COD', 'VNPay', 'Momo') NOT NULL,
    amount DECIMAL(15,0) NOT NULL,
    refunded_amount DECIMAL(15,0) NOT NULL DEFAULT 0 COMMENT 'Running total of Processed refunds; updated in the same transaction as the refund',
    cash_received DECIMAL(15,0) NULL COMMENT 'Counter cash handed over by customer; NULL otherwise',
    change_amount DECIMAL(15,0) NULL COMMENT 'Counter cash returned to customer; NULL otherwise',
    collected_by VARCHAR(36) NULL COMMENT 'External reference to Identity Service; shipper/staff who collected COD cash',
    transaction_id VARCHAR(100) NULL UNIQUE COMMENT 'Gateway transaction id',
    status ENUM('Pending', 'Success', 'Failed', 'Cancelled', 'Expired') NOT NULL DEFAULT 'Pending',
    expires_at DATETIME NULL COMMENT 'Deadline for Pending gateway payments',
    paid_at DATETIME NULL,
    created_by VARCHAR(36) NULL COMMENT 'External reference to Identity Service',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_payments_cashier_shift FOREIGN KEY (cashier_shift_id) REFERENCES Cashier_Shifts(id) ON DELETE RESTRICT,
    CONSTRAINT chk_payments_amount CHECK (amount > 0 AND refunded_amount >= 0 AND refunded_amount <= amount),
    CONSTRAINT chk_payments_cash_fields CHECK (
        (payment_method = 'Cash' AND (
            (cash_received IS NULL AND change_amount IS NULL)
            OR (cash_received IS NOT NULL AND change_amount IS NOT NULL
                AND cash_received >= amount AND change_amount = cash_received - amount)
        ))
        OR (payment_method <> 'Cash' AND cash_received IS NULL AND change_amount IS NULL)
    ),
    CONSTRAINT chk_payments_cash_success CHECK (
        payment_method <> 'Cash' OR status <> 'Success'
        OR (cashier_shift_id IS NOT NULL AND cash_received IS NOT NULL AND change_amount IS NOT NULL)
    ),
    CONSTRAINT chk_payments_cod_success CHECK (
        payment_method <> 'COD' OR status <> 'Success' OR collected_by IS NOT NULL
    ),
    CONSTRAINT chk_payments_gateway_success CHECK (
        payment_method NOT IN ('VNPay', 'Momo') OR status <> 'Success' OR transaction_id IS NOT NULL
    ),
    CONSTRAINT chk_payments_paid_at CHECK (status <> 'Success' OR paid_at IS NOT NULL),
    INDEX idx_payments_cashier_shift (cashier_shift_id),
    INDEX idx_payments_branch_created (branch_id, created_at),
    INDEX idx_payments_status_created (status, created_at),
    INDEX idx_payments_pending_expiry (status, expires_at)
);

-- ============================================================================
-- 06. Inventory Service
-- ============================================================================
-- STOCK FLOW (event-driven, idempotent):
--   order.created   -> create Stock_Reservations, Inventories.reserved_quantity += qty   (prevents oversell)
--   order.completed -> reservation Committed, quantity_in_stock -= qty, reserved_quantity -= qty, log Auto_Deduct
--   order.cancelled -> reservation Released, reserved_quantity -= qty
--   refund (restock_inventory = TRUE) -> log Refund (+)
-- Every Inventory_Logs row carries a unique idempotency_key so a replayed event never double-counts.
-- Stock cannot go below 0. If the business wants negative stock, drop chk_inventories_qty.

-- ============================================================================
-- 07. Engagement Service
-- ============================================================================
-- LOYALTY POLICY: 1 stamp per eligible drink, earned when an order is Completed.
-- Stamps accumulate WITHOUT a cap (no overflow loss). Each 10 stamps = 1 free drink.
-- Redeem = exactly -10 stamps, always tied to the order that contains the reward item.
-- Loyalty_Cards.current_stamps must always equal SUM(Loyalty_Transactions.stamps).
-- A Reversal may push the balance negative ("stamp debt") instead of rewriting history;
-- redeeming requires current_stamps >= 10 (APP RULE).
CREATE DATABASE IF NOT EXISTS smart_cafe_engagement_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE smart_cafe_engagement_db;

CREATE TABLE Loyalty_Cards (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL UNIQUE COMMENT 'External reference to Identity Service',
    current_stamps INT NOT NULL DEFAULT 0,
    lifetime_stamps_earned INT UNSIGNED NOT NULL DEFAULT 0,
    total_redeemed_cups INT UNSIGNED NOT NULL DEFAULT 0 COMMENT 'Lifetime number of free drinks redeemed',
    version INT UNSIGNED NOT NULL DEFAULT 0 COMMENT 'Optimistic lock',
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT chk_loyalty_cards_stamps CHECK (current_stamps >= -100)
);

CREATE TABLE Loyalty_Transactions (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    idempotency_key VARCHAR(64) NULL UNIQUE COMMENT 'Prevents duplicate command/retry processing',
    user_id VARCHAR(36) NOT NULL COMMENT 'External reference to Identity Service',
    order_id BIGINT NULL COMMENT 'External reference to Order Service',
    transaction_type ENUM('Earn', 'Redeem', 'Reversal', 'Adjustment') NOT NULL,
    stamps INT NOT NULL COMMENT 'Signed delta. Earn > 0; Redeem = -10; Reversal opposes the original',
    balance_after INT NOT NULL,
    reversal_of_id BIGINT NULL COMMENT 'The Earn/Redeem transaction being reversed',
    note VARCHAR(255) NULL,
    created_by VARCHAR(36) NULL COMMENT 'External reference to Identity Service; required for manual Adjustment',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    earn_order_id BIGINT GENERATED ALWAYS AS (CASE WHEN transaction_type = 'Earn' THEN order_id ELSE NULL END) STORED,
    redeem_order_id BIGINT GENERATED ALWAYS AS (CASE WHEN transaction_type = 'Redeem' THEN order_id ELSE NULL END) STORED,
    CONSTRAINT fk_loyalty_tx_reversal FOREIGN KEY (reversal_of_id) REFERENCES Loyalty_Transactions(id) ON DELETE RESTRICT,
    CONSTRAINT chk_loyalty_tx_shape CHECK (
        (transaction_type = 'Earn' AND order_id IS NOT NULL AND stamps BETWEEN 1 AND 50 AND reversal_of_id IS NULL)
        OR (transaction_type = 'Redeem' AND order_id IS NOT NULL AND stamps = -10 AND reversal_of_id IS NULL)
        OR (transaction_type = 'Reversal' AND reversal_of_id IS NOT NULL AND stamps <> 0)
        OR (transaction_type = 'Adjustment' AND stamps <> 0 AND created_by IS NOT NULL AND note IS NOT NULL)
    ),
    UNIQUE KEY uq_one_loyalty_earn_per_order (earn_order_id),
    UNIQUE KEY uq_one_loyalty_redeem_per_order (redeem_order_id),
    UNIQUE KEY uq_one_reversal_per_transaction (reversal_of_id),
    INDEX idx_loyalty_tx_user_time (user_id, created_at),
    INDEX idx_loyalty_tx_order (order_id),
    INDEX idx_loyalty_tx_type_time (transaction_type, created_at)
);

-- Customer chatbot MUST filter audience IN ('Customer', 'Both'); staff assistant uses 'Staff' and 'Both'.
-- FULLTEXT works for keyword search; Vietnamese needs app-side tokenization or an embedding/vector store.
CREATE TABLE Knowledge_Base (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    category ENUM('FAQ', 'SOP_Staff', 'Policy', 'Menu_Guide') NOT NULL,
    audience ENUM('Customer', 'Staff', 'Both') NOT NULL DEFAULT 'Customer',
    language CHAR(2) NOT NULL DEFAULT 'vi',
    title VARCHAR(255) NOT NULL,
    content TEXT NOT NULL,
    keywords VARCHAR(255) NULL,
    version INT UNSIGNED NOT NULL DEFAULT 1,
    status ENUM('Active', 'Hidden') NOT NULL DEFAULT 'Active',
    created_by VARCHAR(36) NULL COMMENT 'External reference to Identity Service',
    updated_by VARCHAR(36) NULL COMMENT 'External reference to Identity Service',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT chk_knowledge_staff_only CHECK (category <> 'SOP_Staff' OR audience = 'Staff'),
    INDEX idx_knowledge_lookup (audience, category, status),
    FULLTEXT KEY ft_knowledge_search (title, content, keywords)
);

-- History is kept: each model run inserts new rows; only the newest per target has is_latest = TRUE.
CREATE TABLE AI_Forecasts (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    branch_id INT NOT NULL COMMENT 'External reference to Store Service',
    forecast_type ENUM('Revenue', 'Product_Demand', 'Ingredient_Usage') NOT NULL,
    target_id INT NOT NULL DEFAULT 0 COMMENT '0 for Revenue; Catalog variant_id for Product_Demand; Inventory ingredient_id for Ingredient_Usage',
    forecast_date DATE NOT NULL,
    predicted_value DECIMAL(18,3) NOT NULL COMMENT 'VND for Revenue; units for demand/usage',
    lower_bound DECIMAL(18,3) NULL,
    upper_bound DECIMAL(18,3) NULL,
    actual_value DECIMAL(18,3) NULL COMMENT 'Back-filled later to measure accuracy',
    model_version VARCHAR(50) NOT NULL,
    run_id VARCHAR(36) NOT NULL,
    raw_data JSON,
    is_latest BOOLEAN NOT NULL DEFAULT TRUE,
    generated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    latest_key VARCHAR(80) GENERATED ALWAYS AS (
        CASE WHEN is_latest THEN CONCAT_WS(':', branch_id, forecast_type, target_id, forecast_date) ELSE NULL END
    ) STORED,
    CONSTRAINT chk_ai_forecasts_bounds CHECK (
        (lower_bound IS NULL OR lower_bound <= predicted_value) AND (upper_bound IS NULL OR upper_bound >= predicted_value)
    ),
    UNIQUE KEY uq_ai_forecast_latest (latest_key),
    UNIQUE KEY uq_ai_forecast_run (run_id, branch_id, forecast_type, target_id, forecast_date),
    INDEX idx_ai_forecast_lookup (branch_id, forecast_type, forecast_date)
);

CREATE TABLE AI_Recommendations (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id VARCHAR(36) NOT NULL COMMENT 'External reference to Identity Service',
    branch_id INT NULL COMMENT 'External reference to Store Service',
    product_id INT NOT NULL COMMENT 'External reference to Catalog Service',
    recommendation_type ENUM('Cross_Sell', 'Up_Sell', 'Personalized', 'Trending') NOT NULL,
    confidence_score DECIMAL(5,4) NOT NULL DEFAULT 0.0000,
    reason TEXT NULL,
    model_version VARCHAR(50) NOT NULL,
    expires_at DATETIME NULL,
    clicked_at DATETIME NULL,
    purchased_at DATETIME NULL,
    order_id BIGINT NULL COMMENT 'External reference to Order Service; set when the recommendation converted',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT chk_ai_recommend_score CHECK (confidence_score BETWEEN 0.0000 AND 1.0000),
    CONSTRAINT chk_ai_recommend_conversion CHECK (purchased_at IS NULL OR order_id IS NOT NULL),
    INDEX idx_ai_recommend_user_time (user_id, created_at),
    INDEX idx_ai_recommend_product (product_id),
    INDEX idx_ai_recommend_model (model_version, created_at)
);
